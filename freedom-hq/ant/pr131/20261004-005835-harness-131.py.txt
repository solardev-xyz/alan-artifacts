usage: bench.py <label> <antd-binary> <n> <port>
"""

def wait_ready():
    deadline = time.time() + 120
    pat = "paying=true" if swap == "true" else "settlement"  # unpaid: fire on first peers after chain init
    while time.time() < deadline:
        txt = open(logp).read()

            return True
        if swap != "true" and "retrieval" in txt and "funds" in txt:
            return True
        time.sleep(0.05)
    return False

def get(i, ref):
    path = f"/bzz/{ref}/" if i % 2 == 0 else f"/bytes/{ref}"
    t0 = time.time(); ttfb = None; nbytes = 0; status = None; clen = None; err = None
    try:
        r = urllib.request.urlopen(f"http://127.0.0.1:{port}{path}", timeout=180)

            b = r.read(65536)
            if ttfb is None: ttfb = time.time() - t0

            nbytes += len(b)

import heapq, random
L=1_687_500; R=450_000; PRICE=305_000
def sim(T, F, C, rate, dur=20.0, d=0.06, rtt=0.08, seed=1, applied_trigger=False):
    rnd=random.Random(seed)
    st=dict(bal=0,res=0,sh=0,ong=False,lr=0.0,rif=False,lrc=0.0)
    ev=[]; seq=[0]
    def push(at,kind,data=None): seq[0]+=1; heapq.heappush(ev,(at,seq[0],kind,data))
    cheques=[]; chunks=0; waits=[]
    def cheque_due(now, expected):
        if st['bal']==0 or expected-st['sh']<T or st['bal']-st['sh']<R or st['ong']: return
        rd=int(now-st['lr'])*R
        amt=st['bal']-rd-st['sh']
        if amt<max(F,90_000): return
        st['ong']=True; st['sh']+=amt; push(now+C*rnd.uniform(0.7,1.3),'paid',amt)
    t=0
    while t<dur:
        t+=rnd.expovariate(rate); push(t,'fetch',t)
    push(0.1,'tick')
    while ev:
        now,_,kind,data=heapq.heappop(ev)
        if now>dur+5: break
        if kind=='fetch':
            allow = R if now-st['lr']>=1 else 0
            nxt=st['bal']+st['res']+PRICE
            cheque_due(now, nxt)
            if nxt<=L+allow:
                st['res']+=PRICE; push(now+d*rnd.uniform(0.5,1.5),'apply'); waits.append(now-data)
            else: push(now+0.05,'fetch',data)
        elif kind=='apply':
            st['res']-=PRICE; st['bal']+=PRICE; chunks+=1
            cheque_due(now, st['bal']+st['res'])
        elif kind=='paid':
            st['ong']=False; st['sh']-=data; st['bal']=max(0,st['bal']-data); cheques.append(data)
        elif kind=='tick':
            exp=st['bal']+st['res']-st['sh']; refable=st['bal']-st['sh']
            if not st['rif'] and exp>=675_000 and refable>=R and now>=st['lrc']+1.0:
                st['rif']=True; push(now+rtt,'refreshed')
            push(now+0.1,'tick')
        elif kind=='refreshed':
            el=now-st['lr']; credit=min(st['bal'], int(el)*R)
            st['bal']-=credit; st['lr']=now; st['lrc']=now; st['rif']=False
    waits.sort()
    return len(cheques), (sum(cheques)/len(cheques) if cheques else 0), waits[len(waits)//2], waits[int(len(waits)*.95)], chunks
